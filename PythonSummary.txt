Decorator -- ф-ція яка приймає в параметрах ф-цію і її змінює
			як Middleware на C#
#
	def my_decorator(func):
		def private_decorator():
			print("--My decorator--")
			func()
			print("--End decorator--")
		return private_decorator
	
	@my_decorator		-- @ додаємо декоратор
	def hello_message():
		print("sasdfsdfsdfsdf")
	
	hello_message()		-- invoke ф-ції
#
Результат:
	--My decorator--
	sasdfsdfsdfsdf
	--End decorator--

#
	def view_msg(msg):
		message = msg[0:9]	-- зріз тексту перші 9 символів
		def private_func():		--замкнена ф-ція
			print(message)
		return private_func	--	вказівник на ф-цію
	
	ptr = view_msg("applebanana")
	ptr()

yield(Генератор)	--	послідовнсіть return для foreach
		те саме що Enum
#
	def get_yield_num():
		yield 1
		yield 2
		yield 7
	
	for item in get_yield_num():
		print(item, end="\t")	-- результат '1	2	7'

	def yield_counter(n):
		i=0;
		while i<=n:
			yield i
			i +=2
	
	for item in yield_counter(8):
		print(item, end="\t")	-- 0	2	4 	6	8

| Запис            | Тип                | Приклад            |
| ---------------- | ------------------ | ------------------ |
| `*args`          | tuple              | `(1, 2, 3)`        |
| `**kwargs`       | dict               | `{'name':'Max'}`   |
| `func(*args)`    | розпакування tuple | `func(1,2,3)`      |
| `func(**kwargs)` | розпакування dict  | `func(name='Max')` |

def decorator(func):				--декоратор з параметрами
    def wrapper(*args, **kwargs):
        ...
        return func(*args, **kwargs)
    return wrapper

import os                           -- роботи з папками
from django.conf import settings    --  конфігурація Django
from PIL import Image               -- для роботм з фото
import io                -- для роботм з байтами
import uuid               -- генерація унік імен
from django.core.files.base import ContentFile # Файл який передав користувач

def compress_image(image_field, size=(800,800), quality=85):
  #якщо фото у PNG перетвор у RGB
    image = Image.open(image_field).convert('RGB')
  #Зберіг співвідношення
    image.thumbnail(size, Image.LANCZOS)
  #Генер рандом ім'я
    uid = str(uuid.uuid4())[:10] #10символів назви
    image_name='{}.webp'.format(uid)
  
  #фото зберіг у пам'ять
    output = io.BytesIO()
  #перетвор на webp  
    image.save(output,format='WEBP', quality=quality)
  #зміщ вказівник на початок  
    output.seek(0)

  #отрим саме фото
    resized_image = ContentFile(output.getvalue())
  #повертаємо назву і змін розмір
    return resized_image, image_name
    
def save_custom_image(image, size, folder):
    resized_image, image_name = compress_image(image,size)
    path = os.path.join(folder, image_name)
    full_path = os.path.join(settings.IMAGE_ROOT, path)     
 #створ папку   
    os.makedirs(full_path, exist_ok=True)
    with open(full_path, "wb") as f:
        f.write(resized_image.read())
    return image_name

     class Meta:

    def clean_email(self):
        email = self.cleaned_data.get('email')
        if CustomUser.objects.filter(email=email).exists():
            raise forms.ValidationError("Дана пошта уже зареєстрована")
        return email

    def clean_password2(self):
        password1 = self.cleaned_data.get("password1")
        password2 = self.cleaned_data.get("password2")

        if password1 and password2 and password1 != password2:
            raise forms.ValidationError("Паролі не співпадають.")
        return password2

def user_login(request):
    if request.method == 'POST':
        form = CustomUserLoginForm(data=request.POST)
        if form.is_valid():
            user = authenticate(request, username=form.cleaned_data['email'],
                                password=form.cleaned_data['password'])
            if user is not None:
                login(request, user)
                return redirect('homepage')
    else:
        form = CustomUserLoginForm()
    return render(request, 'login.html', {'form': form})

def user_logout(request):
    logout(request)
    return redirect('homepage')

    from django.db import models 
    from categories.models import Category
    
    class Product(models.Model):
        category = models.ForeignKey(   --посилання на категорію
            Category, on_delete=models.CASCADE, related_name="products"
        )
        name = models.CharField(max_length=255)
        slug = models.SlugField(max_length=255, null=True, blank=True, unique=True)
        description = models.TextField(blank=True)
        price = models.DecimalField(max_digits=10, decimal_places=2)
        created_at = models.DateTimeField(auto_now_add=True)
        updated_at = models.DateTimeField(auto_now=True)
    
        def __str__(self):                              --tostring()
            return f"{self.name} ({self.category.name})"
            
    class ProductImage(models.Model):
        product = models.ForeignKey(Product, on_delete=models.CASCADE, related_name="images", null=True, blank=True)
        image = models.ImageField(upload_to="images/")
        priority = models.PositiveIntegerField(default=0, help_text="0 - Головне фото")     --порядок фото
        created_at = models.DateTimeField(auto_now_add=True)
        updated_at = models.DateTimeField(auto_now=True)
    
        class Meta:
            ordering = ['priority', 'created_at']
        def save(self, *args, **kwargs):
            super().save(*args, **kwargs)
        def __str__(self):
            return f"Фото [{self.priority}] для {self.product.name}"

import {type Control, Controller, type FieldPath, type FieldValues } from "react-hook-form";

type FormInputProps<T extends FieldValues> = {
    control: Control<T>;
    name: FieldPath<T>;
    label: string;
    placeholder?: string;
    type?: React.HTMLInputTypeAttribute;
};

export function FormInput<T extends FieldValues>({
                                                     control,
                                                     name,
                                                     label,
                                                     placeholder,
                                                     type = "text",
                                                 }: FormInputProps<T>) {
    return (
        <Controller
            name={name}
            control={control}
            render={({ field, fieldState }) => (
                <div>
                    <label className="block text-sm font-medium text-slate-700 dark:text-slate-300 mb-1.5">
                        {label}
                    </label>

                    <input
                        {...field}
                        type={type}
                        placeholder={placeholder}
                        className={`
                            w-full px-4 py-2.5 rounded-xl text-sm
                            bg-slate-50 dark:bg-slate-800
                            border ${
                            fieldState.error
                                ? "border-red-400 dark:border-red-500"
                                : "border-slate-200 dark:border-slate-700"
                        }
                            text-slate-900 dark:text-slate-100
                            placeholder:text-slate-400 dark:placeholder:text-slate-500
                            outline-none
                            focus:border-indigo-400 dark:focus:border-indigo-500
                            focus:ring-2 focus:ring-indigo-400/20 dark:focus:ring-indigo-500/20
                            transition-all duration-200
                        `}
                    />

                    {fieldState.error && (
                        <p className="mt-1.5 text-sm text-red-500 dark:text-red-400">
                            {fieldState.error.message}
                        </p>
                    )}
                </div>
            )}
        />
    );
}
