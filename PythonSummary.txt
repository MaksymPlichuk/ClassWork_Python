Decorator -- ф-ція яка приймає в параметрах ф-цію і її змінює
			як Middleware на C#
#
	def my_decorator(func):
		def private_decorator():
			print("--My decorator--")
			func()
			print("--End decorator--")
		return private_decorator
	
	@my_decorator		-- @ додаємо декоратор
	def hello_message():
		print("sasdfsdfsdfsdf")
	
	hello_message()		-- invoke ф-ції
#
Результат:
	--My decorator--
	sasdfsdfsdfsdf
	--End decorator--

#
	def view_msg(msg):
		message = msg[0:9]	-- зріз тексту перші 9 символів
		def private_func():		--замкнена ф-ція
			print(message)
		return private_func	--	вказівник на ф-цію
	
	ptr = view_msg("applebanana")
	ptr()

yield(Генератор)	--	послідовнсіть return для foreach
		те саме що Enum
#
	def get_yield_num():
		yield 1
		yield 2
		yield 7
	
	for item in get_yield_num():
		print(item, end="\t")	-- результат '1	2	7'

	def yield_counter(n):
		i=0;
		while i<=n:
			yield i
			i +=2
	
	for item in yield_counter(8):
		print(item, end="\t")	-- 0	2	4 	6	8

| Запис            | Тип                | Приклад            |
| ---------------- | ------------------ | ------------------ |
| `*args`          | tuple              | `(1, 2, 3)`        |
| `**kwargs`       | dict               | `{'name':'Max'}`   |
| `func(*args)`    | розпакування tuple | `func(1,2,3)`      |
| `func(**kwargs)` | розпакування dict  | `func(name='Max')` |

def decorator(func):				--декоратор з параметрами
    def wrapper(*args, **kwargs):
        ...
        return func(*args, **kwargs)
    return wrapper

import os                           -- роботи з папками
from django.conf import settings    --  конфігурація Django
from PIL import Image               -- для роботм з фото
import io                -- для роботм з байтами
import uuid               -- генерація унік імен
from django.core.files.base import ContentFile # Файл який передав користувач

def compress_image(image_field, size=(800,800), quality=85):
  #якщо фото у PNG перетвор у RGB
    image = Image.open(image_field).convert('RGB')
  #Зберіг співвідношення
    image.thumbnail(size, Image.LANCZOS)
  #Генер рандом ім'я
    uid = str(uuid.uuid4())[:10] #10символів назви
    image_name='{}.webp'.format(uid)
  
  #фото зберіг у пам'ять
    output = io.BytesIO()
  #перетвор на webp  
    image.save(output,format='WEBP', quality=quality)
  #зміщ вказівник на початок  
    output.seek(0)

  #отрим саме фото
    resized_image = ContentFile(output.getvalue())
  #повертаємо назву і змін розмір
    return resized_image, image_name
    
def save_custom_image(image, size, folder):
    resized_image, image_name = compress_image(image,size)
    path = os.path.join(folder, image_name)
    full_path = os.path.join(settings.IMAGE_ROOT, path)     
 #створ папку   
    os.makedirs(full_path, exist_ok=True)
    with open(full_path, "wb") as f:
        f.write(resized_image.read())
    return image_name

     class Meta:

    def clean_email(self):
        email = self.cleaned_data.get('email')
        if CustomUser.objects.filter(email=email).exists():
            raise forms.ValidationError("Дана пошта уже зареєстрована")
        return email

    def clean_password2(self):
        password1 = self.cleaned_data.get("password1")
        password2 = self.cleaned_data.get("password2")

        if password1 and password2 and password1 != password2:
            raise forms.ValidationError("Паролі не співпадають.")
        return password2

def user_login(request):
    if request.method == 'POST':
        form = CustomUserLoginForm(data=request.POST)
        if form.is_valid():
            user = authenticate(request, username=form.cleaned_data['email'],
                                password=form.cleaned_data['password'])
            if user is not None:
                login(request, user)
                return redirect('homepage')
    else:
        form = CustomUserLoginForm()
    return render(request, 'login.html', {'form': form})

def user_logout(request):
    logout(request)
    return redirect('homepage')

#silpo\categories\views.py
    def category_create(request):
        if request.method == 'POST':
            form = CategoryForm(request.POST)
            if form.is_valid():
                try:
                    category = form.save(commit=False)
                    if not category.slug:
                        category.slug = slugify(category.name, allow_unicode=False)
                except Exception as e:
                    messages.error(request, 'Category error', {str(e)})
            else:
                messages.error(request, "Corret errors in Form")
        else:
            form = CategoryForm()
        return render(request, 'categories/category_create.html', {'form':form} )
