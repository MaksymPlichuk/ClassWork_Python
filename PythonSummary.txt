02_DecoratorsClosures

Decorator -- ф-ція яка приймає в параметрах ф-цію і її змінює
			як Middleware на C#
#
	def my_decorator(func):
		def private_decorator():
			print("--My decorator--")
			func()
			print("--End decorator--")
		return private_decorator
	
	@my_decorator		-- @ додаємо декоратор
	def hello_message():
		print("sasdfsdfsdfsdf")
	
	hello_message()		-- invoke ф-ції
#
Результат:
	--My decorator--
	sasdfsdfsdfsdf
	--End decorator--


Closures(замикання)	--	ф-ція замикається всередині і ми її повертаєм
#
	def view_msg(msg):
		message = msg[0:9]	-- зріз тексту перші 9 символів
		def private_func():		--замкнена ф-ція
			print(message)
		return private_func	--	вказівник на ф-цію
	
	ptr = view_msg("applebanana")
	ptr()
#

yield(Генератор)	--	послідовнсіть return для foreach
		те саме що Enum
#
	def get_yield_num():
		yield 1
		yield 2
		yield 7
	
	for item in get_yield_num():
		print(item, end="\t")	-- результат '1	2	7'

	def yield_counter(n):
		i=0;
		while i<=n:
			yield i
			i +=2
	
	for item in yield_counter(8):
		print(item, end="\t")	-- 0	2	4 	6	8
#
| Запис            | Тип                | Приклад            |
| ---------------- | ------------------ | ------------------ |
| `*args`          | tuple              | `(1, 2, 3)`        |
| `**kwargs`       | dict               | `{'name':'Max'}`   |
| `func(*args)`    | розпакування tuple | `func(1,2,3)`      |
| `func(**kwargs)` | розпакування dict  | `func(name='Max')` |

def decorator(func):				--декоратор з параметрами
    def wrapper(*args, **kwargs):
        ...
        return func(*args, **kwargs)
    return wrapper
